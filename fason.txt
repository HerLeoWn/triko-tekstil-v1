// Stok düşme, geri ekleme ve renk toplamları. Giden kayıtlarının kalbi.
import { db } from '../db.js';
import { FULL } from '../stok.js';
import { now, same, anaRenk, guvenliParse, parcaListesi, insertCuval } from '../lib/yardimcilar.js';

// Stokta her rengin toplam seri sayısı: { vanilya: 6, bordo: 3 }
// depoId verilmezse tüm depolar toplanır (biten listesi için)
export function stokSeriByRenk(kod, depoId = null) {
  const map = {};
  db.prepare('SELECT * FROM cuvallar WHERE kod = ? COLLATE NOCASE AND (? IS NULL OR depo_id = ?)')
    .all(kod, depoId, depoId).forEach((r) => {
      parcaListesi(r).forEach((p) => {
        const k = p.renk.trim().toLowerCase();
        map[k] = (map[k] || 0) + p.seri;
      });
    });
  return map;
}

// Verilen renkten `need` seriyi, yalnızca verilen depodaki en eski çuvallardan düşer.
// Çuvalın diğer renkleri yerinde kalır.
export function renktenDus(kod, renk, need, depoId) {
  const rows = db.prepare('SELECT * FROM cuvallar WHERE kod = ? COLLATE NOCASE AND depo_id = ? ORDER BY olusturma')
    .all(kod, depoId);
  const upd = db.prepare('UPDATE cuvallar SET seriler = ?, parcalar = ?, renk = ?, guncelleme = ? WHERE id = ?');
  const del = db.prepare('DELETE FROM cuvallar WHERE id = ?');
  const t = now();
  let kalan = need, cuvalSayisi = 0, cuvalIci = null;

  for (const r of rows) {
    if (kalan <= 0) break;
    const parcalar = parcaListesi(r).map((p) => ({ ...p }));
    const hedef = parcalar.find((p) => same(p.renk, renk) && p.seri > 0);
    if (!hedef) continue;

    const al = Math.min(kalan, hedef.seri);
    hedef.seri -= al;
    kalan -= al;
    cuvalSayisi++;
    cuvalIci ??= r.cuval_ici;

    const kalanParca = parcalar.filter((p) => p.seri > 0);
    const seriler = guvenliParse(r.seriler, []);
    const yeniSeriler = seriler.slice(0, seriler.length - al);

    if (!kalanParca.length || !yeniSeriler.length) del.run(r.id);
    else upd.run(JSON.stringify(yeniSeriler), JSON.stringify(kalanParca), anaRenk(kalanParca), t, r.id);
  }
  return { cuvalSayisi, cuvalIci };
}

// Geri alınan (silinen giden kaydının) seriler yeni çuval olarak, çıktığı depoya eklenir
export function geriEkle(kod, renk, seri, cuvalIci, depoId = 1) {
  const t = now();
  let kalan = seri;
  while (kalan > 0) {
    const k = Math.min(kalan, cuvalIci);
    insertCuval(kod, renk, cuvalIci, [...Array(k)].map(() => [...FULL]), t, [{ renk, seri: k }], depoId);
    kalan -= k;
  }
}
// Giden mallar: listele, ekle, düzenle, sil (hepsi stoktan düşer / geri ekler)
import express from 'express';
import { db } from '../db.js';
import { stokSeriByRenk, renktenDus, geriEkle } from '../services/stokkesim.js';
import { wrap, like, uid, now, logH, depoGecerli, depoAd, guvenliParse } from '../lib/yardimcilar.js';
import { gidenBildirimi } from './mesajlar.js';

const router = express.Router();

// Listele: her kaydın özeti ve çıkış deposu adı ile birlikte
router.get('/api/giden', wrap((req, res) => {
  const q = String(req.query.q || '').trim();
  const rows = db.prepare('SELECT * FROM giden WHERE kod LIKE ? OR renk LIKE ? ORDER BY tarih DESC')
    .all(like(q), like(q));
  const kal = db.prepare('SELECT * FROM giden_kalemler WHERE giden_id = ?');
  res.json(rows.map((g) => {
    const kalemler = kal.all(g.id).map((k) => ({ boyut: k.boyut, cuvalIci: k.cuval_ici, adet: k.adet, stok: !!k.stok }));
    const ozet = kalemler.length
      ? kalemler.map((k) => `${k.adet}×${k.boyut}'li${k.stok ? '' : ' (manuel)'}`).join(' + ')
      : `${g.renk} ${g.seri} seri${g.stok ? '' : ' (manuel)'}`;
    return { ...g, cuvalAdedi: g.cuval_adedi, kalemler, ozet, depo: depoAd(g.depo_id ?? 1) };
  }));
}));

// Yeni giden kaydı(satırları) ekle, seçilen depodan stok düş, muhasebeye bildirim düş
// Giden kaydı: bir kaydetme işlemi = bir nakliyat. Birden fazla model tek seferde girilebilir.
router.post('/api/giden', wrap((req, res) => {
  const ham = Array.isArray(req.body.gruplar) && req.body.gruplar.length
    ? req.body.gruplar
    : [{ kod: req.body.kod, depoId: req.body.depoId, satirlar: req.body.satirlar }];

  const gruplar = ham.map((g) => {
    const kod = String(g.kod ?? '').trim();
    const depoId = depoGecerli(g.depoId);
    const satirlar = (g.satirlar || []).map((s) => ({
      renk: String(s.renk ?? '').trim(),
      seri: parseInt(s.seri, 10),
      stok: s.stok !== false,
    }));
    if (!kod) throw new Error('Kod zorunlu.');
    if (!satirlar.length || satirlar.some((s) => !s.renk || !(s.seri > 0))) {
      throw new Error(`${kod}: her satırda renk ve seri sayısı olmalı.`);
    }
    return { kod, depoId, satirlar };
  });

  db.transaction(() => {
    // Aynı kaydetmedeki tüm satırlar aynı zamanı taşır, fiş bu zamana göre gruplanır
    const t = now();
    const ins = db.prepare(`INSERT INTO giden (id, kod, renk, cuval_adedi, seri, tarih, cuval_ici, stok, depo_id)
                            VALUES (?,?,?,?,?,?,?,?,?)`);
    gruplar.forEach(({ kod, depoId, satirlar }) => {
      const stok = stokSeriByRenk(kod, depoId);
      const ihtiyac = {};
      satirlar.filter((s) => s.stok).forEach((s) => {
        const k = s.renk.toLowerCase();
        ihtiyac[k] = (ihtiyac[k] || 0) + s.seri;
      });
      for (const [k, adet] of Object.entries(ihtiyac)) {
        if (adet > (stok[k] || 0)) {
          throw new Error(`${kod} · ${k}: ${depoAd(depoId)} deposunda yalnızca ${stok[k] || 0} seri var, ${adet} istendi.`);
        }
      }
      satirlar.forEach((s) => {
        let cuvalAdet = 0, cuvalIci = 0;
        if (s.stok) {
          const sonuc = renktenDus(kod, s.renk, s.seri, depoId);
          cuvalAdet = sonuc.cuvalSayisi;
          cuvalIci = sonuc.cuvalIci ?? 0;
        }
        ins.run(uid(), kod, s.renk, cuvalAdet, s.seri, t, cuvalIci, s.stok ? 1 : 0, depoId);
        logH('gitti', kod, s.renk, `${depoAd(depoId)}: ${s.seri} seri${s.stok ? '' : ' (manuel)'}`);
      });
    });
  })();

  res.json({ ok: true });
}));

// Giden kaydını sil: stoktan düşen çuvallar geri eklenir
router.delete('/api/giden/:id', wrap((req, res) => {
  const g = db.prepare('SELECT * FROM giden WHERE id = ?').get(req.params.id);
  if (!g) throw new Error('Giden kaydı bulunamadı.');
  db.transaction(() => {
    const kal = db.prepare('SELECT * FROM giden_kalemler WHERE giden_id = ?').all(g.id);
    if (kal.length) {
      kal.filter((k) => k.stok).forEach((k) => {
        for (let i = 0; i < k.adet; i++) geriEkle(g.kod, g.renk, k.boyut, k.cuval_ici, g.depo_id ?? 1);
      });
    } else if (g.stok && g.cuval_ici) {
      geriEkle(g.kod, g.renk, g.seri, g.cuval_ici, g.depo_id ?? 1);
    }
    db.prepare('DELETE FROM giden WHERE id = ?').run(g.id);
    logH('geri alındı', g.kod, g.renk, `${g.seri} seri kaydı silindi`);
  })();
  res.json({ ok: true });
}));

// Giden kaydını düzenle: eski kaydın stok etkisi geri alınır, yeni bilgiyle yeniden düşülür
router.put('/api/giden/:id', wrap((req, res) => {
  const g = db.prepare('SELECT * FROM giden WHERE id = ?').get(req.params.id);
  if (!g) throw new Error('Giden kaydı bulunamadı.');

  const kod = String(req.body.kod ?? '').trim();
  const renk = String(req.body.renk ?? '').trim();
  const seri = parseInt(req.body.seri, 10);
  const depoId = depoGecerli(req.body.depoId);
  if (!kod || !renk) throw new Error('Kod ve renk zorunlu.');
  if (!(seri > 0)) throw new Error('Seri 1 veya daha büyük olmalı.');

  db.transaction(() => {
    // 1) Eski kaydın stoğa etkisini geri al (silme ile aynı mantık)
    const eskiDepo = g.depo_id ?? 1;
    const kal = db.prepare('SELECT * FROM giden_kalemler WHERE giden_id = ?').all(g.id);
    if (kal.length) {
      kal.filter((k) => k.stok).forEach((k) => {
        for (let i = 0; i < k.adet; i++) geriEkle(g.kod, g.renk, k.boyut, k.cuval_ici, eskiDepo);
      });
    } else if (g.stok && g.cuval_ici) {
      geriEkle(g.kod, g.renk, g.seri, g.cuval_ici, eskiDepo);
    }

    // 2) Yeni kaydı uygula. Giden kayıtları her zaman stoktan düşer.
    const mevcut = stokSeriByRenk(kod, depoId)[renk.toLowerCase()] || 0;
    if (seri > mevcut) {
      throw new Error(`${renk} renginde ${depoAd(depoId)} deposunda yalnızca ${mevcut} seri var, ${seri} istendi.`);
    }
    const sonuc = renktenDus(kod, renk, seri, depoId);

    db.prepare('UPDATE giden SET kod = ?, renk = ?, seri = ?, cuval_adedi = ?, cuval_ici = ?, depo_id = ?, stok = 1 WHERE id = ?')
      .run(kod, renk, seri, sonuc.cuvalSayisi, sonuc.cuvalIci ?? 0, depoId, g.id);
    logH('giden düzenlendi', kod, renk,
      `${depoAd(eskiDepo)} → ${depoAd(depoId)} · ${g.seri} → ${seri} seri`);
  })();
  res.json({ ok: true });
}));

export default router;// Stokta olan ürünlerin açılır listesi. Kod ya da ad yazıldıkça filtrelenir.
// Her üründe hangi depoda ne kadar stok olduğu görünür.
export function StoktakiUrunler({ groups, depolar, arama, onPick }) {
  const q = arama.trim().toLocaleLowerCase('tr-TR');
  const liste = groups.filter((g) =>
    !q
    || g.kod.toLocaleLowerCase('tr-TR').includes(q)
    || (g.ad ?? '').toLocaleLowerCase('tr-TR').includes(q),
  );

  if (!liste.length) {
    return (
      <div className="absolute z-20 mt-1 w-full rounded-lg border border-line bg-white p-3 text-sm text-muted shadow-lift">
        Eşleşen stoklu ürün yok.
      </div>
    );
  }

  return (
    <ul className="absolute z-20 mt-1 max-h-72 w-full overflow-y-auto rounded-lg border border-line bg-white shadow-lift">
      {liste.map((g) => (
        <li key={g.kod} className="border-b border-line last:border-0">
          <button
            type="button"
            className="flex w-full flex-col gap-1.5 px-3 py-2 text-left hover:bg-bg"
            onMouseDown={(e) => { e.preventDefault(); onPick(g); }}
          >
            <span className="flex items-baseline justify-between gap-2">
              <b>{g.kod}</b>
              {g.ad && <span className="truncate text-xs text-muted">{g.ad}</span>}
            </span>
            <span className="flex flex-wrap gap-1">
              {depolar
                .filter((d) => (g.depoSeri?.[d.id] ?? 0) > 0)
                .map((d) => (
                  <span key={d.id} className="rounded-full bg-track px-2 py-0.5 text-[11px]">
                    {d.ad} <b className="tabular-nums">{g.depoSeri[d.id]}</b>
                  </span>
                ))}
            </span>
          </button>
        </li>
      ))}
    </ul>
  );
}
